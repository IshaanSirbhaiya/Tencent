    %% TB3: IB -> sources (Gate 2 live permission check)
    GATE2 ==>|"TB3 · (user, doc) permission question at answer time"| SRC

    %% Platforms -> IB signed webhooks (events only invalidate)
    SRC ==>|"signed webhook · HMAC-SHA256, ±300s, replay-safe; ids only, re-fetch"| WH

    %% TB4: IB -> TokenHub (generation)

    %% TB5: IB -> external ledger-head sink
    LEDGER ==>|"TB5 · ledger head hash {log_id, seq, head, at} (no content)"| SINK

    %% Audit console -> compliance officer
    AUDIT ==>|"metadata + logged reasoned reveal (itself ledgered)"| OFFICER

    %% Ingest embedding boundary note
    IB -.->|"ingest embed: default offline (no chunk leaves host)"| EMB
